// 2. Create types from the service names
type ServiceName = typeof SERVICE_NAMES[number]

interface Service {
  name: ServiceName
  duration: string
  price: string
  steps: string[]
}

interface PackageService {
  name: ServiceName
  isRemovable: boolean
}

interface PackageGroup {
  groupName: string
  selectionType: "exactly-one"
  isRemovable: boolean
  options: { name: ServiceName }[]
}

type PackageItem = PackageService | PackageGroup

interface Package {
  name: string
  price: string | null
  services: PackageItem[]
}
